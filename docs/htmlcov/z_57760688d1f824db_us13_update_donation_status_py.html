1"""US-13 Update Donation Status. 

2 

3Tasks: status form -> view (change status; add/remove inventory when moving to/from Stored; never below 

4zero) -> GUI page -> URL. 

5""" 

6from django import forms 

7from django.contrib import messages 

8from django.core.exceptions import ValidationError

10from django.shortcuts import get_object_or_404, redirect 

11from django.urls import path

13from .common import FORM_FIELDS, render_page 

14from .models import ActivityLog, Donation, apply_donation_inventory_effect 

15from .us04_user_roles import STAFF_ROLES, us04_role_required

18# ---- Form 

19class US13DonationStatusForm(forms.Form): 

20 status = forms.ChoiceField(choices=Donation.STATUS_CHOICES) 

21 

22 

23# ---- GUI 

24TEMPLATE = ( 

25 """<div class="card narrow"><h2>US-13 Update Donation Status</h2> 

26<p>{{ donation.donation_id }} &middot; {{ donation.donor.name }} ({{ donation.blood_group }}) &middot; 

27{{ donation.get_component_display }} &middot; {{ donation.quantity }} unit(s) &middot; {{ donation.blood_bank }}</p> 

28<form method="post" novalidate>{% csrf_token %}""" 

29 + FORM_FIELDS 

30 + """<button class="btn" type="submit">Update status</button> 

31<a class="btn secondary" href="{% url 'us12_view_donation_records' %}">Cancel</a></form></div>""" 

32) 

33 

34 

35# ---- View 

36@us04_role_required(*STAFF_ROLES) 

37def us13_update_donation_status(request, donation_pk): 

38 donation = get_object_or_404(Donation.objects.select_related("donor", "blood_bank"), pk=donation_pk) 

39 form = US13DonationStatusForm(request.POST or None, initial={"status": donation.status}) 

40 if request.method == "POST" and form.is_valid(): 

41 old_status = donation.status 

42 donation.status = form.cleaned_data["status"] 

43 try: 

44 with transaction.atomic(): 

45 donation.save() 

46 apply_donation_inventory_effect(donation, old_status) 

47 except ValidationError as exc: 

48 donation.status = old_status # change was rolled back 

49 form.add_error(None, exc) 

50 else: 

51 ActivityLog.record(request.user, "DONATION_UPDATED", f"{donation.donation_id}: {old_status} -> {donation.status}") 

52 messages.success(request, f"Status of {donation.donation_id} changed to {donation.get_status_display()}.") 

53 return redirect("us12_view_donation_records") 

54 return render_page(request, TEMPLATE, {"form": form, "donation": donation}) 

55 

56
