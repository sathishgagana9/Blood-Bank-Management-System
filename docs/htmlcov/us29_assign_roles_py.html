31@us04_role_required(*ADMIN_ROLES) 

32def us29_assign_roles(request): 

33 selected = request.GET.get("role", "") 

34 users = User.objects.order_by("username") 

35 if selected: 

36 users = users.filter(role=selected) 

37 rows = [] 

38 for u in users: 

39 form = US04UserRoleForm(instance=u, prefix=f"u{u.pk}") 

40 form.fields["role"].widget.attrs["form"] = f"assign-{u.pk}" 

41 rows.append((u, form)) 

42 summary = [(label, User.objects.filter(role=code).count()) for code, label in User.ROLE_CHOICES] 

43 return render_page(request, TEMPLATE, {"rows": rows, "summary": summary, "roles": User.ROLE_CHOICES, "selected": selected})

46@us04_role_required(*ADMIN_ROLES) 

47def us29_update_user_role(request, user_pk): 

48 target = get_object_or_404(User, pk=user_pk) 

49 if request.method == "POST": 

50 form = US04UserRoleForm(request.POST, instance=target, prefix=f"u{target.pk}") 

51 if not form.is_valid(): 

52 messages.error(request, "Invalid role selected.") 

53 elif target == request.user: 

54 messages.error(request, "You cannot change your own role.") 

55 elif target.is_superuser: 

56 messages.error(request, "Superuser accounts are always Administrators.") 

57 else: 

58 old_role = User.objects.get(pk=target.pk).role 

59 new_role = form.cleaned_data["role"] 

60 if old_role == User.ROLE_ADMIN and new_role != User.ROLE_ADMIN and User.objects.filter(role=User.ROLE_ADMIN, is_active=True).count() <= 1: 

61 messages.error(request, "At least one active Administrator must remain.") 

62 else: 

63 form.save() 

64 ActivityLog.record(request.user, "ROLE_ASSIGNED", f"{target.username}: {old_role} -> {new_role}") 

65 messages.success(request, f"{target.username} is now {target.get_role_display()}.") 

66 return redirect("us29_assign_roles")
