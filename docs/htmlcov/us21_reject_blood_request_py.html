18class US21RejectRequestForm(forms.Form): 

19 rejection_reason = forms.CharField(

26 def clean_rejection_reason(self): 

27 reason = self.cleaned_data.get("rejection_reason", "").strip() 

28 if not reason: 

29 raise ValidationError("A reason for rejection must be provided.") 

30 if len(reason) < 5:

103@us04_role_required(*STAFF_ROLES) 

104def us21_pending_requests_list(request): 

105 """Staff view to browse and select pending blood requests to reject.""" 

106 q = request.GET.get("q", "").strip() 

107 requests = BloodRequest.objects.filter(status=BloodRequest.STATUS_PENDING).select_related("blood_bank", "requester") 

108 if q: 

109 requests = requests.filter( 

110 forms.models.Q(request_id__icontains=q) 

111 | forms.models.Q(patient_name__icontains=q) 

112 | forms.models.Q(blood_group__iexact=q) 

113 ) 

114 return render_page(request, TEMPLATE_LIST, {"pending_requests": requests, "q": q})

117@us04_role_required(*STAFF_ROLES) 

118def us21_reject_blood_request(request, pk): 

119 """Staff view to reject a pending blood request with a reason and notify the requester.""" 

120 blood_request = get_object_or_404(BloodRequest.objects.select_related("blood_bank", "requester"), pk=pk) 

121 

122 # Acceptance criteria / test cases: cannot re-reject or modify fulfilled/already resolved requests 

123 if blood_request.status != BloodRequest.STATUS_PENDING: 

124 messages.error(request, f"Request {blood_request.request_id} is already {blood_request.get_status_display()} and cannot be rejected.") 

125 return redirect("us21_pending_requests_list") 

126 

127 form = US21RejectRequestForm(request.POST or None) 

128 if request.method == "POST" and form.is_valid(): 

129 reason = form.cleaned_data["rejection_reason"] 

130 blood_request.status = BloodRequest.STATUS_REJECTED 

131 blood_request.rejection_reason = reason 

132 blood_request.handled_by = request.user 

133 blood_request.save() 

134 

135 # Notify requester (US-21 / US-25) 

136 if blood_request.requester: 

137 Notification.objects.create( 

138 user=blood_request.requester, 

139 category="REQUEST_STATUS", 

140 message=f"Your blood request {blood_request.request_id} has been Rejected. Reason: {reason}", 

141 link="/requests/my/", 

142 ) 

143 

144 # Audit log entry (US-33) 

145 ActivityLog.record( 

146 request.user, 

147 "REJECT_REQUEST", 

148 f"Rejected blood request {blood_request.request_id}. Reason: {reason}", 

149 ) 

150 

151 messages.success(request, f"Blood request {blood_request.request_id} has been rejected and the requester notified.") 

152 return redirect("us21_pending_requests_list") 

153 

154 return render_page(request, TEMPLATE_REJECT, {"blood_request": blood_request, "form": form})
