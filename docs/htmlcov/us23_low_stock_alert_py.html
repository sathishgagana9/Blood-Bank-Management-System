1"""US-23 Low Stock Alert. 

2 

3Tasks: automatic check after every stock change (called from BloodInventory.adjust) -> notify all staff and 

4administrators (US-24 inbox) -> GUI page listing everything currently low -> URL. 

5The limit is LOW_STOCK_THRESHOLD in settings.py (default 5 units). 

6""" 

7from django.conf import settings 

8from django.urls import path, reverse 

9 

10from .common import render_page 

11from .models import BloodInventory, Notification, User 

12from .us04_user_roles import STAFF_ROLES, us04_role_required 

13 

14 

15def us23_threshold(): 

16 return getattr(settings, "LOW_STOCK_THRESHOLD", 5) 

17 

18 

19def us23_check_low_stock(row, old_quantity): 

20 """Send an alert when stock went DOWN and is now at or below the threshold.""" 

21 if row.quantity < old_quantity and row.quantity <= us23_threshold(): 

22 staff = User.objects.filter(role__in=STAFF_ROLES, is_active=True) 

23 Notification.send( 

24 staff, "LOW_STOCK", 

25 f"Low stock: {row.blood_bank} has only {row.quantity} unit(s) of {row.blood_group} {row.get_component_display()}.", 

26 link=reverse("us23_low_stock_alert"), 

27 ) 

28 

29 

30# ---- GUI 

31TEMPLATE = """<div class="card"><h2>US-23 Low Stock Alerts</h2> 

32<p>Showing stock at or below <strong>{{ threshold }}</strong> unit(s).</p> 

33<table><tr><th>Blood bank</th><th>Blood group</th><th>Component</th><th>Units left</th><th>Last updated</th><th></th></tr> 

34{% for r in rows %} 

35<tr><td>{{ r.blood_bank }}</td><td>{{ r.blood_group }}</td><td>{{ r.get_component_display }}</td> 

36<td style="color:#b00020;font-weight:700">{{ r.quantity }}</td><td>{{ r.updated_at|date:"d M Y H:i" }}</td> 

37<td><a href="{% url 'us15_update_blood_inventory' %}">Update stock</a></td></tr> 

38{% empty %}<tr><td colspan="6">No low-stock items. 🎉</td></tr>{% endfor %} 

39</table></div>""" 

40 

41 

42# ---- View 

43@us04_role_required(*STAFF_ROLES) 

44def us23_low_stock_alert(request): 

45 rows = BloodInventory.objects.select_related("blood_bank").filter( 

46 quantity__lte=us23_threshold(), blood_bank__is_active=True 

47 ).order_by("quantity") 

48 return render_page(request, TEMPLATE, {"rows": rows, "threshold": us23_threshold()}) 

49 

50 

51URLS = [path("alerts/low-stock/", us23_low_stock_alert, name="us23_low_stock_alert")]
