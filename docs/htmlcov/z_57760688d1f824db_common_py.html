83def render_page(request, body, context=None): 

84 """Wrap a story's GUI template (`body`) in the common page frame and render it.""" 

85 context = dict(context or {}) 

86 if request.user.is_authenticated: # unread count shown next to "Notifications" (US-24) 

87 context["unread_count"] = request.user.notifications.filter(is_read=False).count() 

88 template = engines["django"].from_string(PAGE_START + body + PAGE_END) 

89 return HttpResponse(template.render(context, request))
