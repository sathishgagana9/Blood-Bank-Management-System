1"""US-24 New Request Notification. 

2 

3Tasks: notify every staff member and administrator when a blood request is submitted (called from US-18) 

4-> notification inbox page (also used by US-23, US-25, US-26) with unread count in the menu -> URLs. 

5""" 

6from django.contrib import messages 

7from django.shortcuts import redirect 

8from django.urls import path, reverse 

9 

10from .common import render_page 

11from .models import Notification, User 

12from .us04_user_roles import ALL_ROLES, STAFF_ROLES, us04_role_required 

13 

14 

15def us24_notify_new_request(blood_request): 

16 staff = User.objects.filter(role__in=STAFF_ROLES, is_active=True) 

17 Notification.send( 

18 staff, "NEW_REQUEST", 

19 f"New {blood_request.get_urgency_display().upper()} blood request {blood_request.request_id}: {blood_request.quantity} unit(s) of {blood_request.blood_group} " 

20 f"{blood_request.get_component_display()} at {blood_request.blood_bank} (needed by {blood_request.required_by}).", 

21 link=reverse("us19_view_blood_requests"), 

22 ) 

23 

24 

25# ---- GUI 

26TEMPLATE = """<div class="card"><h2>US-24 Notifications</h2> 

27<form method="post" action="{% url 'us24_mark_notifications_read' %}">{% csrf_token %}<button class="btn" type="submit">Mark all as read</button></form> 

28<table><tr><th>When</th><th>Type</th><th>Message</th></tr> 

29{% for n in notifications %} 

30<tr{% if not n.is_read %} style="font-weight:700;background:#fff7e6"{% endif %}><td>{{ n.created_at|date:"d M Y H:i" }}</td><td>{{ n.get_category_display }}</td> 

31<td>{% if n.link %}<a href="{{ n.link }}">{{ n.message }}</a>{% else %}{{ n.message }}{% endif %}</td></tr> 

32{% empty %}<tr><td colspan="3">No notifications yet.</td></tr>{% endfor %} 

33</table></div>""" 

34 

35 

36# ---- Views 

37@us04_role_required(*ALL_ROLES) 

38def us24_view_notifications(request): 

39 return render_page(request, TEMPLATE, {"notifications": request.user.notifications.all()[:100]}) 

40 

41 

42@us04_role_required(*ALL_ROLES) 

43def us24_mark_notifications_read(request): 

44 if request.method == "POST": 

45 request.user.notifications.filter(is_read=False).update(is_read=True) 

46 messages.success(request, "All notifications marked as read.") 

47 return redirect("us24_view_notifications") 

48 

49 

50URLS = [ 

51 path("notifications/", us24_view_notifications, name="us24_view_notifications"), 

52 path("notifications/read/", us24_mark_notifications_read, name="us24_mark_notifications_read"), 

53]
