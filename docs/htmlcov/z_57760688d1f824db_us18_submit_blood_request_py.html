19class US18SubmitRequestForm(forms.ModelForm): 

20 blood_bank = forms.ModelChoiceField(queryset=BloodBank.objects.filter(is_active=True)) 

21 

22 class Meta: 

23 model = BloodRequest 

24 fields = ("patient_name", "blood_bank", "blood_group", "component", "quantity", "urgency", "contact", "required_by", "notes") 

25 widgets = {

30 def clean_required_by(self): 

31 date = self.cleaned_data["required_by"] 

32 if date < timezone.localdate(): 

33 raise forms.ValidationError("The required-by date cannot be in the past.")

47# ---- View 

48@us04_role_required(*ALL_ROLES) 

49def us18_submit_blood_request(request): 

50 initial = {k: request.GET[k] for k in ("blood_bank", "blood_group", "component") if request.GET.get(k)} 

51 initial.setdefault("contact", request.user.phone) 

52 form = US18SubmitRequestForm(request.POST or None, initial=initial) 

53 if request.method == "POST" and form.is_valid(): 

54 blood_request = form.save(commit=False) 

55 blood_request.requester = request.user 

56 blood_request.save() 

57 ActivityLog.record(request.user, "REQUEST_SUBMITTED", f"{blood_request.request_id}: {blood_request.quantity} x {blood_request.blood_group}") 

58 us24_notify_new_request(blood_request) 

59 stock = BloodInventory.objects.filter( 

60 blood_bank=blood_request.blood_bank, blood_group=blood_request.blood_group, component=blood_request.component 

61 ).first() 

62 messages.success(request, f"Blood request {blood_request.request_id} submitted. You will be notified of its status.") 

63 if not stock or stock.quantity < blood_request.quantity: 

64 messages.warning(request, "Note: this blood bank currently has less stock than you requested; staff will review it.") 

65 return redirect("us19_view_blood_requests") 

66 return render_page(request, TEMPLATE, {"form": form})
